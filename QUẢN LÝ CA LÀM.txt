-- =====================================================================
-- CSDL: HỆ THỐNG QUẢN LÝ NHÂN SỰ VÀ CA LÀM CHO QUÁN CÀ PHÊ
-- DBMS: PostgreSQL (chạy trên pgAdmin)
-- LƯU Ý: Hãy chắc chắn bạn đang mở Query Tool bên trong database 'qlcl'
-- =====================================================================

-- NHÓM: HỆ THỐNG
CREATE TABLE cua_hang (
    id SERIAL PRIMARY KEY,
    ten_cua_hang VARCHAR(100) NOT NULL,
    dia_chi VARCHAR(255),
    vi_do DECIMAL(9,6),
    kinh_do DECIMAL(9,6),
    trang_thai BOOLEAN DEFAULT TRUE
);

CREATE TABLE vi_tri (
    id SERIAL PRIMARY KEY,
    ten_vi_tri VARCHAR(100) NOT NULL,
    mo_ta VARCHAR(255)
);

CREATE TABLE vai_tro (
    id SERIAL PRIMARY KEY,
    ten_vai_tro VARCHAR(50) NOT NULL,
    mo_ta VARCHAR(255)
);

-- NHÓM: HỒ SƠ NHÂN SỰ
CREATE TABLE nhan_vien (
    id SERIAL PRIMARY KEY,
    ma_nhan_vien VARCHAR(50) UNIQUE,
    ma_cua_hang INT,
    ma_vi_tri INT,
    ho_ten VARCHAR(100) NOT NULL,
    so_dien_thoai VARCHAR(20),
    email VARCHAR(255),
    dia_chi VARCHAR(255),
    ngay_sinh DATE,
    gioi_tinh VARCHAR(10),
    tinh_trang_hon_nhan VARCHAR(50),
    so_nhan_khau INT,
    trang_thai VARCHAR(20) DEFAULT 'dang_lam',
    ngay_tao TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_nv_cua_hang FOREIGN KEY (ma_cua_hang) REFERENCES cua_hang(id),
    CONSTRAINT fk_nv_vi_tri   FOREIGN KEY (ma_vi_tri)   REFERENCES vi_tri(id)
);

CREATE TABLE hop_dong (
    id SERIAL PRIMARY KEY,
    ma_nhan_vien INT NOT NULL,
    loai_hop_dong VARCHAR(20),
    ngay_bat_dau DATE,
    ngay_ket_thuc DATE,
    muc_luong DECIMAL(12,2),
    ma_bang_luong VARCHAR(25),
    trang_thai VARCHAR(20),
    ngay_tao TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_hd_nv FOREIGN KEY (ma_nhan_vien) REFERENCES nhan_vien(id)
);

-- NHÓM: XÁC THỰC & PHÂN QUYỀN
CREATE TABLE tai_khoan (
    id SERIAL PRIMARY KEY,
    ma_nv INT NOT NULL,
    ma_vai_tro INT,
    ten_dang_nhap VARCHAR(255) NOT NULL UNIQUE,
    mat_khau VARCHAR(255) NOT NULL,
    is_active BOOLEAN DEFAULT TRUE,
    da_dang_nhap_lan_cuoi TIMESTAMP,
    lockout_until TIMESTAMP,
    so_lan_dang_nhap_sai INT DEFAULT 0,
    ngay_tao TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_tk_nv        FOREIGN KEY (ma_nv)        REFERENCES nhan_vien(id),
    CONSTRAINT fk_tk_vai_tro   FOREIGN KEY (ma_vai_tro)   REFERENCES vai_tro(id)
);

CREATE TABLE lich_su_dang_nhap (
    id SERIAL PRIMARY KEY,
    ma_tai_khoan INT NOT NULL,
    thoi_gian_dang_nhap TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    dia_chi_ip VARCHAR(45),
    trang_thai VARCHAR(20),
    CONSTRAINT fk_lsdn_tk FOREIGN KEY (ma_tai_khoan) REFERENCES tai_khoan(id)
);

-- NHÓM: ĐĂNG KÝ & XẾP CA
CREATE TABLE loai_ca (
    id SERIAL PRIMARY KEY,
    ma_loai_ca VARCHAR(50) UNIQUE,
    mo_ta TEXT,
    trang_thai BOOLEAN DEFAULT TRUE
);

CREATE TABLE lich_trinh (
    id SERIAL PRIMARY KEY,
    ma_nv INT NOT NULL,
    ngay_bat_dau DATE,
    ngay_ket_thuc DATE,
    trang_thai VARCHAR(20),
    da_duoc_duyet DATE,
    CONSTRAINT fk_lt_nv FOREIGN KEY (ma_nv) REFERENCES nhan_vien(id)
);

CREATE TABLE dang_ky_ca (
    id SERIAL PRIMARY KEY,
    ma_nv INT NOT NULL,
    ma_ca INT NOT NULL,
    ngay_dang_ky DATE,
    trang_thai VARCHAR(20),
    da_duyet BOOLEAN DEFAULT FALSE,
    CONSTRAINT fk_dkc_nv FOREIGN KEY (ma_nv) REFERENCES nhan_vien(id),
    CONSTRAINT fk_dkc_ca FOREIGN KEY (ma_ca) REFERENCES loai_ca(id)
);

CREATE TABLE phan_cong_ca (
    id SERIAL PRIMARY KEY,
    ma_lich_trinh INT,
    ma_nv INT NOT NULL,
    ma_ca INT NOT NULL,
    bat_dau_thuc_te TIMESTAMP,
    ket_thuc_thuc_te TIMESTAMP,
    trang_thai VARCHAR(20),
    ghi_chu VARCHAR(255),
    CONSTRAINT fk_pcc_lt FOREIGN KEY (ma_lich_trinh) REFERENCES lich_trinh(id),
    CONSTRAINT fk_pcc_nv FOREIGN KEY (ma_nv)         REFERENCES nhan_vien(id),
    CONSTRAINT fk_pcc_ca FOREIGN KEY (ma_ca)         REFERENCES loai_ca(id)
);

CREATE TABLE cham_cong (
    id SERIAL PRIMARY KEY,
    ma_phan_cong INT NOT NULL,
    check_in_lan_1 TIMESTAMP,
    check_out_lan_1 TIMESTAMP,
    check_in_lan_2 TIMESTAMP,
    check_out_lan_2 TIMESTAMP,
    check_in_lan_3 TIMESTAMP,
    check_out_lan_3 TIMESTAMP,
    so_gio_di_trong INT,
    tac_vu_phu VARCHAR(25),
    gio_phut_nghi INT,
    tong_gio_lam DECIMAL(6,2),
    thang VARCHAR(7),
    nam INT,
    CONSTRAINT fk_cc_pcc FOREIGN KEY (ma_phan_cong) REFERENCES phan_cong_ca(id)
);

-- NHÓM: THANH TOÁN & BÁO CÁO
CREATE TABLE yeu_cau_thanh_toan (
    id SERIAL PRIMARY KEY,
    ma_nv INT NOT NULL,
    so_tien DECIMAL(10,2),
    mo_ta VARCHAR(255),
    trang_thai VARCHAR(20),
    da_duyet BOOLEAN DEFAULT FALSE,
    CONSTRAINT fk_yctt_nv FOREIGN KEY (ma_nv) REFERENCES nhan_vien(id)
);

CREATE TABLE muc_thu_chi (
    id SERIAL PRIMARY KEY,
    ma_phan_cong INT,
    ma_nhan_vien INT NOT NULL,
    loai VARCHAR(20),
    so_tien DECIMAL(10,2),
    ly_do VARCHAR(255),
    trang_thai VARCHAR(20),
    da_tao TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_mtc_pcc FOREIGN KEY (ma_phan_cong) REFERENCES phan_cong_ca(id),
    CONSTRAINT fk_mtc_nv  FOREIGN KEY (ma_nhan_vien)  REFERENCES nhan_vien(id)
);

CREATE TABLE phieu_luong (
    id SERIAL PRIMARY KEY,
    ma_phan_cong INT,
    cong_huong DECIMAL(10,2),
    tong_thu_nhap DECIMAL(10,2),
    luong_co_ban DECIMAL(10,2),
    thanh_tien DECIMAL(10,2),
    thang INT,
    nam INT,
    CONSTRAINT fk_pl_pcc FOREIGN KEY (ma_phan_cong) REFERENCES phan_cong_ca(id)
);

CREATE TABLE xuat_bao_cao (
    id SERIAL PRIMARY KEY,
    ma_phieu_luong INT,
    ky_luong VARCHAR(20),
    loai VARCHAR(10),
    file_url VARCHAR(255),
    da_xuat TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_xbc_pl FOREIGN KEY (ma_phieu_luong) REFERENCES phieu_luong(id)
);

CREATE TABLE bao_cao_export (
    id SERIAL PRIMARY KEY,
    ma_phan_cong INT,
    thang INT,
    nam INT,
    pham_vi VARCHAR(20),
    loai_bao_cao VARCHAR(50),
    ten_file VARCHAR(255),
    da_tao TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_bce_pcc FOREIGN KEY (ma_phan_cong) REFERENCES phan_cong_ca(id)
);

-- NHÓM: NHẬT KÝ HỆ THỐNG / AUDIT
CREATE TABLE thong_bao (
    id SERIAL PRIMARY KEY,
    ma_tai_khoan INT NOT NULL,
    tieu_de TEXT,
    noi_dung VARCHAR(500),
    loai_thong_bao VARCHAR(50),
    da_doc BOOLEAN DEFAULT FALSE,
    ngay_tao TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_tb_tk FOREIGN KEY (ma_tai_khoan) REFERENCES tai_khoan(id)
);

CREATE TABLE nhat_ky_he_thong (
    id SERIAL PRIMARY KEY,
    ma_tai_khoan INT NOT NULL,
    ma_cua_hang INT,
    hanh_dong VARCHAR(100),
    thoi_gian TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ly_do TEXT,
    trang_thai VARCHAR(255),
    CONSTRAINT fk_nkht_tk FOREIGN KEY (ma_tai_khoan) REFERENCES tai_khoan(id),
    CONSTRAINT fk_nkht_ch FOREIGN KEY (ma_cua_hang)  REFERENCES cua_hang(id)
);

CREATE TABLE nhat_ky_audit (
    id SERIAL PRIMARY KEY,
    ma_tai_khoan INT NOT NULL,
    ma_nhat_ky_he_thong INT,
    hanh_dong TEXT,
    gia_tri_cu TEXT,
    gia_tri_moi TEXT,
    ly_do VARCHAR(255),
    thoi_gian TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_nka_tk   FOREIGN KEY (ma_tai_khoan)        REFERENCES tai_khoan(id),
    CONSTRAINT fk_nka_nkht FOREIGN KEY (ma_nhat_ky_he_thong) REFERENCES nhat_ky_he_thong(id)
);

CREATE TABLE audit_log (
    id SERIAL PRIMARY KEY,
    user_account_id INT NOT NULL,
    ma_nhat_ky_he_thong INT UNIQUE,
    table_name VARCHAR(100),
    record_id INT,
    old_value TEXT,
    new_value TEXT,
    reason VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_al_tk   FOREIGN KEY (user_account_id)     REFERENCES tai_khoan(id),
    CONSTRAINT fk_al_nkht FOREIGN KEY (ma_nhat_ky_he_thong) REFERENCES nhat_ky_he_thong(id)
);

-- INDEX PHỤ TRỢ
CREATE INDEX idx_phan_cong_ca_nv    ON phan_cong_ca(ma_nv);
CREATE INDEX idx_cham_cong_thang_nam ON cham_cong(thang, nam);
CREATE INDEX idx_phieu_luong_thang_nam ON phieu_luong(thang, nam);
CREATE INDEX idx_dang_ky_ca_ngay    ON dang_ky_ca(ngay_dang_ky);
